# 量測核可 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 量測先進「待核可」,老師在核可頁用圓餅分組選組後,以該組平均登記;公開摘要/趨勢/地圖/匯出只用核可值。

**Architecture:** 「量測紀錄」第 11 欄記核可編號;「核可紀錄」分頁那一列是 commit point(有效編號集合決定哪些量測已核可)。分組規則 `groupValues` 前後端各一份、同案例測試。公開 `summary` 只列有核可的樹並加 `pending`;`history` 每次核可一點。新頁 `approve.html`。

**Tech Stack:** 純 HTML/JS ES modules、Google Apps Script(ES5 風格)、vitest。

**Spec:** `docs/superpowers/specs/2026-09-25-approval-design.md`

## Global Constraints
- 分組:樹高 `u=Math.round(h*100)`、樹圍 `u=Math.round(g*10)`;組距整數 `10|50|100`;`k=Math.floor(u/w)`;平均 `Math.round(sum/n)/scale`;每位學生(學號)只算量測時間最新一筆,無學號的列各自算一人;預設人數最多、同票取 k 最小並 `tie:true`。
- 待核可 = `parseMeasurementRow` 有效,且第 11 欄空白或不在「有效」核可編號集合。
- 核可順序:先標第 11 欄,最後寫核可紀錄(commit)。撤銷只改狀態。
- 公開輸出逐欄白名單,不含 `@`、姓名、學號、紀錄編號;公開 GET 只用 `getSheetByName`;快取鍵 `tree-summary-v2`、`tree-history-v2:`。
- Code.gs 參數:`Number.isInteger` 驗證;map 用 `Object.create(null)`;寫入文字 `sanitizeCellText`。
- 前端不用 `innerHTML`;不新增 npm 套件;不 commit(使用者未要求);改 Code.gs 要提醒部署。
- 測試:`npx.cmd vitest run`。快取:學生頁相關檔案變動 → `CACHE_NAME` +1(v19→v20),`phonePolish.test.js` 同步。

```js
// 核可頁的圓餅分組。規則與 apps-script/Code.gs 的 groupValues 逐字對應(改一邊要改另一邊)。
// 換成整數單位(樹高公分、樹圍公釐)再分組與平均,避免 12.3/0.1 = 122.999… 的浮點誤差。
export function groupValues(records, { field, scale, width }) {
  const latest = new Map();
  records.forEach((rec, i) => {
    if (!(Number(rec[field]) > 0)) return;
    const who = rec.classNo ? `s:${rec.classNo}` : `r:${i}`;
    const prev = latest.get(who);
    if (!prev || String(rec.at) > String(prev.at)) latest.set(who, rec);
  });
  const byK = new Map();
  for (const rec of latest.values()) {
    const u = Math.round(Number(rec[field]) * scale);
    const k = Math.floor(u / width);
    const g = byK.get(k) || { k, sum: 0, count: 0, latestAt: '' };
    g.sum += u;
    g.count += 1;
    if (String(rec.at) > g.latestAt) g.latestAt = String(rec.at);
    byK.set(k, g);
  }
  const groups = [...byK.values()]
    .sort((a, b) => a.k - b.k)
    .map((g) => ({ k: g.k, lo: (g.k * width) / scale, hi: ((g.k + 1) * width) / scale, count: g.count, mean: Math.round(g.sum / g.count) / scale, latestAt: g.latestAt }));
  let best = null;
  let tie = false;
  for (const g of groups) {
    if (!best || g.count > best.count) { best = g; tie = false; } else if (g.count === best.count) tie = true;
  }
  return { groups, defaultK: best ? best.k : null, tie };
}

export function groupLabel(group, unit, decimals) {
  return `${group.lo.toFixed(decimals)}–${group.hi.toFixed(decimals)} ${unit}`;
}
```

```js
// ---------------------------------------------------------------------------
// 量測核可(老師專用)
// ---------------------------------------------------------------------------
var APPROVAL_WIDTHS = [10, 50, 100]; // 樹高:公分;樹圍:公釐

function recordsSheet() {
  return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_RECORDS);
}

function measurementKey(m) {
  return m.rid + '|' + m.height + '|' + (m.girth === null ? '' : m.girth);
}

/** 該樹所有待核可的有效量測,附 1-based 列號與 key。 */
function pendingFor(treeId, approvals) {
  var validIds = validApprovalIds(approvals);
  var rows = recordsSheet().getDataRange().getValues();
  var out = [];
  for (var i = 1; i < rows.length; i++) {
    var m = parseMeasurementRow(rows[i]);
    if (m && m.no === treeId && isPending(m, validIds)) {
      m.row = i + 1;
      m.key = measurementKey(m);
      out.push(m);
    }
  }
  return out;
}

function latestValidApproval(approvals, treeId, field) {
  var best = null;
  for (var i = 0; i < approvals.length; i++) {
    var a = approvals[i];
    if (a.valid && a.no === treeId && (!best || a[field] > best[field])) best = a;
  }
  return best;
}

function cleanTreeId(raw) {
  return String(raw === null || raw === undefined ? '' : raw).trim();
}

function sameKeys(a, b) {
  if (!Array.isArray(b) || a.length !== b.length) return false;
  var x = a.slice().sort();
  var y = b.map(String).sort();
  for (var i = 0; i < x.length; i++) if (x[i] !== y[i]) return false;
  return true;
}

function approvalBatch(data) {
  var treeId = cleanTreeId(data.treeId);
  if (treeId === '') return errorOutput('VALIDATION_FAILED', '缺少樹號');
  var approvals = readApprovals();
  var last = latestValidApproval(approvals, treeId, 'approvedAt');
  return jsonOutput({
    status: 'ok',
    treeId: treeId,
    records: pendingFor(treeId, approvals).map(function (m) {
      return { key: m.key, classNo: m.classNo, at: m.at, height: m.height, girth: m.girth };
    }),
    last: last ? { id: last.id, at: last.measuredAt, height: last.height, girth: last.girth } : null,
  });
}

function pickGroup(result, k) {
  for (var i = 0; i < result.groups.length; i++) if (result.groups[i].k === k) return result.groups[i];
  return null;
}

function approvalLabel(g, unit, decimals) {
  return g.lo.toFixed(decimals) + '–' + g.hi.toFixed(decimals) + ' ' + unit;
}

function approvalApprove(data, email) {
  var treeId = cleanTreeId(data.treeId);
  var bad = errorOutput('VALIDATION_FAILED', '組距或組別不正確,請重新整理');
  if (treeId === '' || APPROVAL_WIDTHS.indexOf(data.heightWidth) < 0 || !Number.isInteger(data.heightK)) return bad;
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var sheet = recordsSheet();
    var batch = pendingFor(treeId, readApprovals());
    if (!sameKeys(batch.map(function (m) { return m.key; }), data.keys)) {
      return errorOutput('BATCH_CHANGED', '有新的量測或別的老師剛處理過,已重新載入');
    }
    var h = pickGroup(groupValues(batch, 'height', 100, data.heightWidth), data.heightK);
    if (!h || !(h.mean > 0) || !isFinite(h.mean)) return bad;
    var girthGroups = null;
    var g = null;
    if (data.girthK === null || data.girthK === undefined) {
      if (groupValues(batch, 'girth', 10, 100).groups.length > 0) return bad; // 有樹圍資料就必須選一組
    } else {
      if (APPROVAL_WIDTHS.indexOf(data.girthWidth) < 0 || !Number.isInteger(data.girthK)) return bad;
      girthGroups = groupValues(batch, 'girth', 10, data.girthWidth);
      g = pickGroup(girthGroups, data.girthK);
      if (!g || !(g.mean > 0) || !isFinite(g.mean)) return bad;
    }

    // 1. 標記(寫前確認第 10 欄沒被手動改動)
    if (sheet.getMaxColumns() < COLUMN_APPROVAL_ID) sheet.insertColumnsAfter(COLUMN_APPROVAL_ID - 1, 1);
    sheet.getRange(1, COLUMN_APPROVAL_ID).setValue('核可編號');
    var id = Utilities.getUuid();
    for (var i = 0; i < batch.length; i++) {
      var rid = sheet.getRange(batch[i].row, COLUMN_CLIENT_RECORD_ID, 1, 1).getValues()[0][0];
      if (String(rid) !== batch[i].rid) return errorOutput('BATCH_CHANGED', '試算表剛被改動,已重新載入');
      sheet.getRange(batch[i].row, COLUMN_APPROVAL_ID).setValue(id);
    }

    // 2. commit:寫入核可紀錄這一列,核可才生效
    var ap = getOrCreateSheet(SHEET_NAME_APPROVALS, APPROVAL_HEADER);
    var newRow = ap.getLastRow() + 1;
    ap.getRange(newRow, 1, 1, APPROVAL_HEADER.length).setNumberFormat('@');
    ap.getRange(newRow, 1, 1, APPROVAL_HEADER.length).setValues([[
      id, sanitizeCellText(treeId), h.latestAt, nowIso(), sanitizeCellText(email),
      h.mean, approvalLabel(h, 'm', 1), h.count,
      g ? g.mean : '', g ? approvalLabel(g, 'cm', 0) : '', g ? g.count : '',
      APPROVAL_VALID, '',
    ]]);
  } finally {
    lock.releaseLock();

function approvalUndo(data, email) {

    if (!target.valid) return jsonOutput({ status: 'ok', id: id, duplicate: true });
    var latest = latestValidApproval(approvals, treeId, 'approvedAt');
    if (latest.id !== id) return errorOutput('BATCH_CHANGED', '這棵樹剛有新的核可,已重新載入');

      .getRange(target.row, 12, 1, 2)
      .setValues([['已撤銷 by ' + sanitizeCellText(email), nowIso()]]);
  } finally {
    lock.releaseLock();
  }
  invalidateCachesFor(treeId);
  return jsonOutput({ status: 'ok', id: id });
}
```
